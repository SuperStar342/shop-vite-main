/* 使用 序列对象  ,根据单据号产生新的 ID  
  
一般用于新增单据时,单据号还没有确定.  
如新增一订单, 订单号还未填写或写自动产生, 用户可能先加附件, 那么这些附件就不能依赖订单号,  
可这样处理: 用户点新增按钮后,立即获取new id , 后面加个附件依赖于这个id  
   
注意: 不能在事务中调用本sp  
    不保证id连续  
   占用多个id的话,需要自己额外处理  
  
Date  Modifier  Remark  
--------------------------------------------------  
*/  
CREATE PROCEDURE dbo.Sp_Sys_GetNewID_BySeq_V2
    @ReceiptCode VARCHAR(50), 
    @Nums INT = 1, 
    @NewID INT OUTPUT,
    @RtnTblResult BIT = 1,
    @Debug BIT = 0,
    @DBTable VARCHAR(50) = ''
AS
BEGIN
    DECLARE 
        @ts TIMESTAMP, 
        @msg NVARCHAR(500), 
        @oldid INT,
        @startIden INT,
        @SeqName VARCHAR(300),
        @next INT,
        @SeqExisted BIT
		,@sql NVARCHAR(max);

    -- 确保参数正确
    SELECT 
        @ReceiptCode = ISNULL(@ReceiptCode, ''),
        @DBTable = ISNULL(@DBTable, '');
	IF @Nums=0
		SELECT @Nums=1

    IF LEN(@DBTable) = 0 OR @DBTable IS NULL
        SET @DBTable = @ReceiptCode; -- 默认表名为空时使用单据编码

    SELECT 
        @SeqName ='iden_so_'+@DBTable,
        @SeqExisted = 
		CASE WHEN EXISTS(SELECT * FROM sys.sequences WITH(nolock) WHERE name = @SeqName) THEN 1 ELSE 0 END;

    IF @SeqExisted=0 -- 序列不存在
    BEGIN
        SET @NewID = 1;
		SELECT @oldid = COALESCE((SELECT TOP 1 fNextID FROM dbo.t_SysM_ReceiptNewID WITH(nolock) WHERE [fReceiptCode] = @DBTable ORDER BY [fNextID]), 0);
		IF @DBTable='t_POMD_OrdSubPackageMst'--t_POMD_OrdSubPackageMst
		BEGIN
					
				DECLARE @MaxOrdSubPackageMstID INT 
				SELECT @MaxOrdSubPackageMstID=MAX(fID)
				FROM t_POMD_OrdSubPackageMst WITH(NOLOCK)
				SELECT @MaxOrdSubPackageMstID=ISNULL(@MaxOrdSubPackageMstID,0)
				select @oldid= @MaxOrdSubPackageMstID +1
		END 

		IF @oldid>@NewID SELECT @NewID=@oldid

		IF NOT EXISTS(select 1 from sys.sequences a WITH(nolock) WHERE a.name=@seqName )    
	    BEGIN
			EXEC Sp_Sys_CreateSequence @seqName=@seqName ,@StartValue=@NewID
			--WAITFOR DELAY '00:00:01'
	    END
    END



    SELECT @ts = 0X

	IF @Nums =1
	BEGIN
		set @sql='
				select @pSno= next value for ['+@SeqName+']
			'; 
		Exec Sp_executesql
			@stmt=@sql,
			@params=N'@pSno [numeric](18,0) OUTPUT',
			@pSno=@newID OUTPUT
    END
	ELSE 
	BEGIN 
		DECLARE @Sno_variant SQL_VARIANT

		EXEC sys.sp_sequence_get_range @sequence_name=@seqName,@range_size=@Nums
					,@range_first_value=@Sno_variant OUTPUT
		SELECT @newID=CONVERT([NUMERIC](18, 0),@Sno_variant)
	END

	IF @RtnTblResult = 1   
    BEGIN  
        SELECT  fid = @newID  , fid2 =@newID+@Nums-1 ,  
                fnextid = @NewID + 1 , fts = @ts  
    END  

END;
